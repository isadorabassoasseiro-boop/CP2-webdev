Number - Converte em um número inteiro ou decimal e retorna NaN caso a conversão não seja possível.
console.log(Number("42")); // 42 (número inteiro)
console.log(Number("42.99")); // 42.99 (mantém os decimais)
console.log(Number("42px")); // NaN (não consegue converter)
console.log(Number(" 42 ")); // 42 (ignora espaços)
console.log(Number("")); // 0 (string vazia vira zero)
console.log(Number(true)); // 1 (true → 1, false → 0)

parteInt - Converte em um número inteiro e para no primeiro caractere inválido.
console.log(parseInt("42")); // 42
console.log(parseInt("42.99")); // 42 (descarta a parte decimal)
console.log(parseInt("42px")); // 42 (para ao encontrar um caractere inválido)
console.log(parseInt(" 42 ")); // 42 (ignora espaços)
console.log(parseInt("")); // NaN (não consegue converter)
console.log(parseInt("0xF", 16)); // 15 (interpreta como hexadecimal)
console.log(parseInt("101", 2)); // 5 (interpreta como binário)

parseFloat - Semelhante ao parseInt, porém converte mantendo o valor decimal.
console.log(parseFloat("42")); // 42
console.log(parseFloat("42.99")); // 42.99 (mantém a parte decimal)
console.log(parseFloat("42.99px")); // 42.99 (para ao encontrar um caractere inválido)
console.log(parseFloat(" 42.5 ")); // 42.5 (ignora espaços)
console.log(parseFloat("3.14e2")); // 314 (interpreta notação científica)
console.log(parseFloat("abc42")); // NaN (não começa com número)
